  function goToStep(n){
    document.querySelectorAll('.booking-form').forEach(f=>f.classList.remove('active'));
    document.querySelectorAll('.step-btn').forEach(b=>b.classList.remove('active'));
    const f=document.getElementById('step-'+n);
    const b=document.getElementById('step-btn-'+n);
    if(f)f.classList.add('active');
    if(b)b.classList.add('active');
  }

  function nextStep(cur){
    if(cur===1&&!bk.service){alert('Please select a treatment to continue.');return;}
    if(cur===2){
      bk.date=document.getElementById('booking-date').value;
      if(!bk.date||!bk.time){alert('Please select a date and time slot.');return;}
    }
    if(cur===3){
      const fn=document.getElementById('fname').value.trim();
      const em=document.getElementById('email').value.trim();
      if(!fn||!em){alert('Please fill in your name and email.');return;}
      bk.name=fn+' '+document.getElementById('lname').value.trim();
      bk.email=em;
      updateSummary();
    }
    goToStep(cur+1);
  }
  function prevStep(cur){goToStep(cur-1);}

  function updateSummary(){
    const dep=Math.round(bk.price*.15);
    document.getElementById('sum-service').textContent=bk.service;
    document.getElementById('sum-datetime').textContent=(bk.date||'—')+' · '+(bk.time||'—');
    document.getElementById('sum-name').textContent=bk.name;
    document.getElementById('sum-total').textContent='RM '+bk.price.toLocaleString();
    const depSec=document.getElementById('deposit-section');
    const credSec=document.getElementById('credit-confirm-section');
    const depLbl=document.getElementById('deposit-label');
    const sumDep=document.getElementById('sum-deposit');
    const confirmBtn=document.getElementById('confirm-btn');
    if(bk.useCredit){
      depSec.style.display='none';credSec.style.display='block';
      depLbl.textContent='Payment';sumDep.textContent='Credit Package';
      confirmBtn.textContent='Confirm with Credit Package ✓';
    } else {
      depSec.style.display='block';credSec.style.display='none';
      depLbl.textContent='15% Deposit Due Now';
      sumDep.textContent='RM '+dep.toLocaleString();
      document.getElementById('deposit-display').textContent='RM '+dep.toLocaleString();
      confirmBtn.textContent='Pay Deposit & Confirm ✓';
    }
  }

  function fmtCard(inp){
    let v=inp.value.replace(/\D/g,'');
    inp.value=v.match(/.{1,4}/g)?.join(' ')||v;
  }

  function submitBooking(){
    document.querySelectorAll('.booking-form').forEach(f=>f.classList.remove('active'));
    document.querySelectorAll('.step-btn').forEach(b=>b.classList.remove('active'));
    const msg=bk.useCredit
      ?`Your session has been deducted from your credit package.<br>Valerie will WhatsApp you shortly to confirm. See you at Shinju Skin!`
      :`Your booking is secured and your 15% deposit has been received.<br>Valerie will WhatsApp you shortly to confirm. See you at Shinju Skin!`;
    document.getElementById('success-msg').innerHTML=msg;
    document.getElementById('success-screen').style.display='block';
  }

  function resetBooking(){
    bk={category:'',service:'',price:0,time:'',date:'',name:'',email:'',useCredit:false};
    document.getElementById('success-screen').style.display='none';
    document.getElementById('treatment-select-wrap').style.display='none';
    document.getElementById('credit-toggle').checked=false;
    document.getElementById('credit-info').classList.remove('show');
    document.getElementById('credit-fields').classList.remove('show');
    document.querySelectorAll('.service-opt').forEach(o=>o.classList.remove('selected'));
    document.querySelectorAll('.time-slot').forEach(t=>t.classList.remove('selected'));
    goToStep(1);
  }

  function openLb(el, cap){

    lbImg.alt = cap;
    document.getElementById('lb-cap').textContent = cap;
    document.getElementById('lightbox').classList.add('open');
  }
